#   [birthday]            — birthday wishes
#   [anniversary]         — wedding/relationship anniversary
#   [custom:work]         — work anniversary / promotion

#   [custom:graduation]   — graduation
#   [custom:get_well]     — get well soon
#   [custom]              — general fallback (uses {occasion_label})

# Rules:
#   • Lines starting with # are comments — ignored at runtime
#   • A blank line between templates is fine; the [tag] line starts each one
#   • Add as many templates as you like — one is picked at random
#   • For anniversaries without a partner_name, {partner_name} is removed
